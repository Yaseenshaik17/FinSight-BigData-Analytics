# Import 10,000 customer JSON documents into MongoDB finsight.customers collection
mongoimport --db finsight --collection customers --drop --file "/home/clouduser/Downloads/Bigdata-Dataset/src-data/noveacrest_customers.json" --jsonArray


// Switch to project database
use finsight

// Requirement R1: Create compound index on customerId and segment for query optimization
db.customers.createIndex({ customerId: 1, segment: 1 })

// Requirement R2: Run aggregation pipeline to validate customer counts per segment
db.customers.aggregate([
  {
    $group: {
      _id: "$segment",
      count: { $sum: 1 }
    }
  },
  {
    $sort: { count: -1 }
  }
])
